<body>
    <div class="container">
        <h1>Какой ты шинопровод? 🤩</h1>
        <p>Пройди тест и узнай!</p>
        <button onclick="startTest()">Начать</button>
        <div id="test" style="display:none;"></div>
        <div id="result" style="display:none;"></div>
    </div>
    
    <script>
        const questions = [
            { text: "Какую температуру ты предпочитаешь?", options: ["Жарче, чем в сауне! 🔥", "Люблю стабильность 🌡️", "Чем холоднее, тем лучше! ❄️", "Мне всё равно, лишь бы работать! ⚙️"] },
            { text: "Какие погодные условия тебе нравятся?", options: ["Пыль, влага, грязь – моя стихия! 🌪️", "Я создан для идеальных условий ☀️", "Люблю снег, дождь, ураган! 🌨️🌪️", "Главное, чтобы не залило! ☂️"] },
            { text: "Ты готов работать в агрессивной среде?", options: ["Химикаты, кислоты? Легко! 🧪", "Предпочитаю чистоту и порядок 🏡", "Можно немного испытаний 🏗️", "Я бронированный! 🦾"] },
            { text: "Какую нагрузку ты выдержишь?", options: ["Максимум мощности! ⚡", "Средний уровень – надежность! 🔩", "Легкие нагрузки, но с запасом! 🔋", "Главное – баланс! ⚖️"] }
        ];

        let answers = [];
        let currentQuestion = 0;

        function startTest() {
            document.querySelector("#test").style.display = "block";
            document.querySelector("button").style.display = "none";
            showQuestion();
        }

        function showQuestion() {
            if (currentQuestion >= questions.length) {
                showResult();
                return;
            }
            let q = questions[currentQuestion];
            let testDiv = document.querySelector("#test");
            testDiv.innerHTML = `<p>${q.text}</p>`;
            q.options.forEach((option, index) => {
                let btn = document.createElement("button");
                btn.innerText = option;
                btn.onclick = function () {
                    answers.push(index);
                    currentQuestion++;
                    showQuestion();
                };
                testDiv.appendChild(btn);
            });
        }

 function showResult() {
            let resultText = "";
            if (answers[0] === 0) {
                resultText = "Ты – троллейный шинопровод! Надежный, как дедовский топор, проверенный временем и готовый работать даже в условиях апокалипсиса! ⚡";
            } else if (answers[0] === 2) {
                resultText = "Ты – монотроллейный шинопровод! Тебя не испугать холодом, снегом и ледяным кофе – ты справишься даже в условиях Антарктиды! ❄️";
            } else if (answers[2] === 0) {
                resultText = "Ты – изолированный рельс U12! Ты выживаешь там, где другие ломаются, выдерживаешь химатаки и атмосферу запущенного гаража! 🦾";
            } else {
                resultText = "Ты – гибкий шинопровод! Готов адаптироваться к любым ситуациям, даже если тебя свернут в узел – ты всё равно будешь работать! 🔄";
            }
            resultText += "<br><br>Нужен шинопровод? Обращайтесь к нам – мы профессионалы и поможем с выбором! <a href='https://tetikaprom.ru' target='_blank'>Наш сайт</a>.";
            document.querySelector("#test").style.display = "none";
            document.querySelector("#result").style.display = "block";
            document.querySelector("#result").innerHTML = `<h2>Результат:</h2><p>${resultText}</p>`;
        }
    </script>
